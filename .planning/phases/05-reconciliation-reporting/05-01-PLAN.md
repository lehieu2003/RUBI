Create directory src/app/api/orders/[id]/reconcile/ and file route.ts.

The reconcile endpoint atomically:
1. Validates session (ACCOUNTANT or ADMIN only)
2. Verifies order is DELIVERED (return 409 if already COMPLETED)
3. Updates each OrderItem.actualQty
4. Updates order status to COMPLETED
5. Creates CustomerDebtLedger DEBT entry
6. Upserts Inventory record (decrement quantityOnHand)

```typescript
import { NextRequest, NextResponse } from 'next/server';
import { prisma } from '@/lib/db';
import { getSession } from '@/lib/auth';
import { z } from 'zod';

const ReconcileSchema = z.object({
  items: z.array(z.object({
    id: z.string(),
    actualQty: z.number().int().min(0),
  })).min(1),
});

export async function PATCH(
  request: NextRequest,
  { params }: { params: Promise<{ id: string }> }
) {
  try {
    // --- Session auth ---
    const session = await getSession();
    if (!session) {
      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
    }
    if (session.role !== 'ACCOUNTANT' && session.role !== 'ADMIN') {
      return NextResponse.json({ error: 'Forbidden' }, { status: 403 });
    }

    const { id } = await params;
    const body = await request.json();
    const parsed = ReconcileSchema.safeParse(body);
    if (!parsed.success) {
      return NextResponse.json({ error: 'Invalid body. Required: items[{id, actualQty}]' }, { status: 400 });
    }

    // --- Check order is DELIVERED ---
    const order = await prisma.order.findUnique({
      where: { id },
      include: { orderItems: true, customer: true },
    });
    if (!order) {
      return NextResponse.json({ error: 'Order not found' }, { status: 404 });
    }
    if (order.status !== 'DELIVERED') {
      return NextResponse.json(
        { error: `Order is ${order.status}, cannot reconcile. Only DELIVERED orders can be reconciled.`, status: order.status },
        { status: 409 }
      );
    }

    // --- Atomic transaction: reconcile + inventory deduction + debt entry ---
    await prisma.$transaction(async (tx) => {
      // 1. Update actualQty for each line item
      for (const item of parsed.data.items) {
        await tx.orderItem.update({
          where: { id: item.id },
          data: { actualQty: item.actualQty },
        });
      }

      // 2. Update order status to COMPLETED
      const updatedOrder = await tx.order.update({
        where: { id },
        data: { status: 'COMPLETED' },
      });

      // 3. Calculate delivered total from actualQty * unitPrice
      const deliveredTotal = parsed.data.items.reduce((sum, itemInput) => {
        const orderItem = order.orderItems.find(oi => oi.id === itemInput.id);
        if (!orderItem) return sum;
        return sum + (itemInput.actualQty * orderItem.unitPrice.toNumber());
      }, 0);

      // 4. Create DEBT ledger entry
      if (deliveredTotal > 0) {
        await tx.customerDebtLedger.create({
          data: {
            customerId: order.customerId,
            type: 'DEBT',
            amount: deliveredTotal,
            notes: `Order ${order.id} delivered`,
          },
        });
      }

      // 5. Deduct inventory for each product (upsert daily record)
      const today = new Date();
      today.setHours(0, 0, 0, 0);

      for (const itemInput of parsed.data.items) {
        const orderItem = order.orderItems.find(oi => oi.id === itemInput.id);
        if (!orderItem || itemInput.actualQty === 0) continue;

        await tx.inventory.upsert({
          where: {
            productId_date: { productId: orderItem.productId, date: today },
          },
          create: {
            productId: orderItem.productId,
            date: today,
            quantityOnHand: -itemInput.actualQty, // Negative = deduction from stock
          },
          update: {
            quantityOnHand: { decrement: itemInput.actualQty },
          },
        });
      }
    });

    return NextResponse.json({
      success: true,
      orderId: id,
      status: 'COMPLETED',
    });
  } catch (error) {
    console.error('[PATCH /api/orders/[id]/reconcile]', error);
    return NextResponse.json({ error: 'Reconciliation failed' }, { status: 500 });
  }
}
```

```typescript
import { NextRequest, NextResponse } from 'next/server';
import { prisma } from '@/lib/db';
import { getSession } from '@/lib/auth';

export async function GET(request: NextRequest) {
  try {
    // --- Session auth ---
    const session = await getSession();
    if (!session) {
      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
    }
    if (session.role !== 'ACCOUNTANT' && session.role !== 'ADMIN') {
      return NextResponse.json({ error: 'Forbidden' }, { status: 403 });
    }

    const customers = await prisma.customer.findMany({
      include: {
        debtLedger: true,
      },
    });

    const debtReport = customers.map((customer) => {
      const totalDebt = customer.debtLedger
        .filter(e => e.type === 'DEBT')
        .reduce((sum, e) => sum + e.amount.toNumber(), 0);
      const totalPayment = customer.debtLedger
        .filter(e => e.type === 'PAYMENT')
        .reduce((sum, e) => sum + e.amount.toNumber(), 0);
      const outstanding = totalPayment - totalDebt; // Positive = customer owes

      // Find last payment date
      const paymentEntries = customer.debtLedger.filter(e => e.type === 'PAYMENT');
      const lastPaymentDate = paymentEntries.length > 0
        ? paymentEntries.sort((a, b) => b.createdAt.getTime() - a.createdAt.getTime())[0].createdAt
        : null;

      return {
        customerId: customer.id,
        customerName: customer.name,
        outstandingDebt: outstanding,
        lastPaymentDate,
      };
    });

    // Sort by outstanding debt descending (highest debt first)
    debtReport.sort((a, b) => b.outstandingDebt - a.outstandingDebt);

    return NextResponse.json(debtReport);
  } catch (error) {
    console.error('[GET /api/reports/debt]', error);
    return NextResponse.json({ error: 'Failed to fetch debt report' }, { status: 500 });
  }
}
```

```typescript
import { NextRequest, NextResponse } from 'next/server';
import { prisma } from '@/lib/db';
import { getSession } from '@/lib/auth';
import { z } from 'zod';

const ShellDepositSchema = z.object({
  customerId: z.string(),
  shellsOut: z.number().int().min(0).optional(),
  shellsIn: z.number().int().min(0).optional(),
  notes: z.string().optional(),
});

export async function GET(request: NextRequest) {
  try {
    const session = await getSession();
    if (!session) {
      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
    }
    if (session.role !== 'ACCOUNTANT' && session.role !== 'ADMIN') {
      return NextResponse.json({ error: 'Forbidden' }, { status: 403 });
    }

    const customers = await prisma.customer.findMany({
      include: {
        shellDeposit: true,
      },
    });

    const deposits = customers
      .map((customer) => {
        const deposit = customer.shellDeposit;
        const netDepositOwed = deposit
          ? (deposit.shellsOut - deposit.shellsIn) * deposit.depositPerShell.toNumber()
          : 0;

        return {
          customerId: customer.id,
          customerName: customer.name,
          shellsOut: deposit?.shellsOut ?? 0,
          shellsIn: deposit?.shellsIn ?? 0,
          depositPerShell: deposit?.depositPerShell?.toNumber() ?? 50000,
          netDepositOwed,
        };
      })
      .filter(d => d.shellsOut > 0 || d.shellsIn > 0); // Only show customers with deposit activity

    // Sort by netDepositOwed descending (most owed first)
    deposits.sort((a, b) => b.netDepositOwed - a.netDepositOwed);

    return NextResponse.json(deposits);
  } catch (error) {
    console.error('[GET /api/deposits/shells]', error);
    return NextResponse.json({ error: 'Failed to fetch deposit data' }, { status: 500 });
  }
}

export async function PATCH(request: NextRequest) {
  try {
    const session = await getSession();
    if (!session) {
      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
    }
    if (session.role !== 'ACCOUNTANT' && session.role !== 'ADMIN') {
      return NextResponse.json({ error: 'Forbidden' }, { status: 403 });
    }

    const body = await request.json();
    const parsed = ShellDepositSchema.safeParse(body);
    if (!parsed.success) {
      return NextResponse.json({ error: 'Invalid body' }, { status: 400 });
    }

    const { customerId, shellsOut, shellsIn } = parsed.data;

    // Verify customer exists
    const customer = await prisma.customer.findUnique({ where: { id: customerId } });
    if (!customer) {
      return NextResponse.json({ error: 'Customer not found' }, { status: 404 });
    }

    await prisma.customerShellDeposit.upsert({
      where: { customerId },
      create: {
        customerId,
        shellsOut: shellsOut ?? 0,
        shellsIn: shellsIn ?? 0,
        depositPerShell: 50000, // Default 50,000 VND per D-09
      },
      update: {
        shellsOut: { increment: shellsOut ?? 0 },
        shellsIn: { increment: shellsIn ?? 0 },
      },
    });

    return NextResponse.json({
      success: true,
      customerId,
      shellsOut: shellsOut ?? 0,
      shellsIn: shellsIn ?? 0,
    });
  } catch (error) {
    console.error('[PATCH /api/deposits/shells]', error);
    return NextResponse.json({ error: 'Failed to update deposit' }, { status: 500 });
  }
}
```
