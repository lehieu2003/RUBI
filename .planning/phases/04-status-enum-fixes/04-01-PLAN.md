**Rewrite the PATCH handler** in `src/app/api/orders/[id]/confirm/route.ts` to add Bearer token auth middleware, order ownership check, and improve error handling.

The new handler must:
1. Extract `Authorization` header, verify it starts with `Bearer `
2. Look up driver by token using `prisma.driver.findUnique({ where: { token } })`
3. Return 401 if token is invalid
4. Fetch order and verify `order.driverId === driver.id`
5. Return 403 if driver does not own the order
6. Proceed with DELIVERED status update + GPS save

**Full replacement** of the PATCH export:
```typescript
export async function PATCH(
  request: NextRequest,
  { params }: { params: Promise<{ id: string }> }
) {
  try {
    const { id } = await params;

    // --- Bearer token auth check ---
    const authHeader = request.headers.get('authorization');
    if (!authHeader?.startsWith('Bearer ')) {
      return NextResponse.json({ error: 'Missing or invalid Authorization header' }, { status: 401 });
    }
    const token = authHeader.slice(7);

    const driver = await prisma.driver.findUnique({
      where: { token },
    });
    if (!driver) {
      return NextResponse.json({ error: 'Invalid token' }, { status: 401 });
    }

    // --- Request body parsing ---
    const body = await request.json();
    const parsed = ConfirmSchema.safeParse(body);

    if (!parsed.success) {
      return NextResponse.json(
        { error: 'Invalid request. Required: gps { lat, lng } or gps: null' },
        { status: 400 }
      );
    }

    // --- Order ownership check ---
    const order = await prisma.order.findUnique({ where: { id } });
    if (!order) {
      return NextResponse.json({ error: 'Order not found' }, { status: 404 });
    }
    if (order.driverId !== driver.id) {
      return NextResponse.json({ error: 'Not authorized to confirm this order' }, { status: 403 });
    }

    // --- Update order to DELIVERED with GPS coordinates ---
    const gpsValue = parsed.data.gps == null ? Prisma.JsonNull : parsed.data.gps;
    const updated = await prisma.order.update({
      where: { id },
      data: {
        status: 'DELIVERED',
        deliveryGps: gpsValue,
      },
    });

    return NextResponse.json({
      success: true,
      orderId: updated.id,
      status: updated.status,
      deliveryGps: updated.deliveryGps,
    });
  } catch (error) {
    // Handle Prisma not found error specifically
    if (error instanceof Error && error.name === 'NotFoundError') {
      return NextResponse.json({ error: 'Order not found' }, { status: 404 });
    }
    console.error('[PATCH /api/orders/[id]/confirm]', error);
    return NextResponse.json({ error: 'Confirmation failed' }, { status: 500 });
  }
}
```

Keep the `ConfirmSchema` and imports as they are — only replace the PATCH handler body.
