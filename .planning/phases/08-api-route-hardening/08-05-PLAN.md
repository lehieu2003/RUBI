```typescript
import type { Subscription, SubscriptionPlan, SubscriptionStatus } from '@prisma/client';

export type VietQRData = {
  bankName: string;
  accountNumber: string;
  accountHolder: string;
  amount?: number;
  note?: string;
};

// Format for VietQR-compatible string (manual payment reference)
export function formatVietQRString(data: VietQRData): string {
  const parts = [
    data.bankName,
    data.accountNumber,
    data.accountHolder,
    data.amount ? data.amount.toString() : '',
    data.note || '',
  ];
  return parts.join('|');
}

// Generate payment reference note
export function formatPaymentNote(factoryName: string, subscriptionId: string): string {
  return `Thanh toan IceFlow ${factoryName} - ${subscriptionId.slice(0, 8)}`;
}

// VietQR display template for UI rendering
export interface VietQRDisplay {
  bankName: string;
  accountNumber: string;
  accountHolder: string;
  formattedAccount: string;  // Masked: 1234xxxx5678
  amount?: number;
  note: string;
  qrString: string;
}

export function createVietQRDisplay(
  bankName: string,
  accountNumber: string,
  accountHolder: string,
  amount?: number,
  note?: string
): VietQRDisplay {
  // Mask account number for display: show first 4 and last 4 digits
  const masked = accountNumber.length > 8
    ? accountNumber.slice(0, 4) + 'xxxx' + accountNumber.slice(-4)
    : accountNumber;

  const qrData: VietQRData = {
    bankName,
    accountNumber,
    accountHolder,
    amount,
    note,
  };

  return {
    bankName,
    accountNumber,
    accountHolder,
    formattedAccount: masked,
    amount,
    note: note || '',
    qrString: formatVietQRString(qrData),
  };
}
```

This creates the utility file with VietQR data types, formatting function, and display helper.
