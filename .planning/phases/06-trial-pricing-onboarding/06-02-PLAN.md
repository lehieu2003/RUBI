```typescript
'use client';

import type { Subscription } from '@prisma/client';
import { useState } from 'react';

interface PaymentRecorderProps {
  subscription: Subscription;
  factoryId: string;
}

export function PaymentRecorder({ subscription, factoryId }: PaymentRecorderProps) {
  const [amount, setAmount] = useState('');
  const [loading, setLoading] = useState(false);
  const [message, setMessage] = useState<{ type: 'success' | 'error'; text: string } | null>(null);

  const expectedAmount = Math.max(0, (subscription.driverCount || 0) - 1) * 150000;

  const handleConfirm = async () => {
    setLoading(true);
    setMessage(null);
    
    try {
      const res = await fetch('/api/payments/confirm', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          subscriptionId: subscription.id,
          amount: parseInt(amount || String(expectedAmount)),
          paidAt: new Date().toISOString(),
        }),
      });
      
      if (res.ok) {
        setMessage({ type: 'success', text: 'Đã xác nhận thanh toán! Đăng ký đã được gia hạn.' });
        setAmount('');
        // Optionally refresh subscription data
      } else {
        const data = await res.json();
        setMessage({ type: 'error', text: data.error || 'Có lỗi xảy ra' });
      }
    } catch {
      setMessage({ type: 'error', text: 'Không thể kết nối server' });
    } finally {
      setLoading(false);
    }
  };

  return (
    <div className="bg-white rounded-lg border p-4">
      <h2 className="text-lg font-semibold mb-4">Xác nhận thanh toán</h2>
      
      <div className="space-y-4">
        <div>
          <label className="block text-sm font-medium text-gray-700 mb-1">
            Số tiền (VND)
          </label>
          <input
            type="number"
            value={amount || expectedAmount}
            onChange={e => setAmount(e.target.value)}
            className="w-full px-3 py-2 border rounded-lg"
            placeholder={String(expectedAmount)}
          />
          <p className="text-xs text-gray-500 mt-1">
            Số tiền mong đợi: {expectedAmount.toLocaleString('vi-VN')} VND
          </p>
        </div>
        
        <button
          onClick={handleConfirm}
          disabled={loading}
          className="w-full px-4 py-2 bg-blue-600 text-white rounded-lg font-medium hover:bg-blue-700 disabled:opacity-50"
        >
          {loading ? 'Đang xử lý...' : 'Xác nhận thanh toán'}
        </button>
        
        {message && (
          <div className={`p-3 rounded-lg text-sm ${message.type === 'success' ? 'bg-green-50 text-green-700' : 'bg-red-50 text-red-700'}`}>
            {message.text}
          </div>
        )}
      </div>
    </div>
  );
}
```

```typescript
import { NextRequest, NextResponse } from 'next/server';
import { PrismaClient } from '@prisma/client';
import { getServerSession } from '@/lib/auth';
import { getFactorySubscription, isSubscriptionActive } from '@/lib/subscription-status';

const prisma = new PrismaClient();

export async function POST(req: NextRequest) {
  try {
    const session = await getServerSession();
    if (!session) {
      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
    }
    if (session.role !== 'ADMIN') {
      return NextResponse.json({ error: 'Admin only' }, { status: 403 });
    }

    const { subscriptionId, amount, paidAt } = await req.json();

    if (!subscriptionId || typeof amount !== 'number' || amount <= 0) {
      return NextResponse.json({ error: 'Invalid input' }, { status: 400 });
    }

    // Verify subscription belongs to this factory
    const subscription = await prisma.subscription.findUnique({
      where: { id: subscriptionId },
    });

    if (!subscription || subscription.factoryId !== session.factoryId) {
      return NextResponse.json({ error: 'Subscription not found' }, { status: 404 });
    }

    // Update subscription: extend +30 days, set to PAID and ACTIVE
    const updatedSubscription = await prisma.subscription.update({
      where: { id: subscriptionId },
      data: {
        planType: 'PAID',
        status: 'ACTIVE',
        endDate: new Date(Date.now() + 30 * 24 * 60 * 60 * 1000), // +30 days
        driverCount: subscription.driverCount, // preserve driver count
      },
    });

    return NextResponse.json({
      success: true,
      subscription: updatedSubscription,
    });
  } catch (error) {
    console.error('Payment confirm error:', error);
    return NextResponse.json({ error: 'Internal server error' }, { status: 500 });
  }
}
```

```typescript
'use client';

import { useState, useRef } from 'react';
import { useRouter } from 'next/navigation';
import Papa from 'papaparse';

type Step = 'upload' | 'mapping' | 'preview' | 'importing' | 'done';

interface CustomerRow {
  code: string;
  name: string;
  phone: string;
  address: string;
}

export default function OnboardingPage() {
  const router = useRouter();
  const [step, setStep] = useState<Step>('upload');
  const [csvData, setCsvData] = useState<string[][]>([]);
  const [headers, setHeaders] = useState<string[]>([]);
  const [mapping, setMapping] = useState<{ code: number; name: number; phone: number; address: number }>({ code: 0, name: 1, phone: 2, address: 3 });
  const [customers, setCustomers] = useState<CustomerRow[]>([]);
  const [importResult, setImportResult] = useState<{ imported: number; skipped: number } | null>(null);

  const handleFileUpload = (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0];
    if (!file) return;
    
    Papa.parse(file, {
      complete: (results) => {
        const data = results.data as string[][];
        if (data.length > 0) {
          setHeaders(data[0]);
          setCsvData(data.slice(1).filter(row => row.some(cell => cell.trim())));
        }
        setStep('mapping');
      },
    });
  };

  const handleMapping = () => {
    const mapped: CustomerRow[] = csvData.map(row => ({
      code: row[mapping.code]?.trim() || '',
      name: row[mapping.name]?.trim() || '',
      phone: row[mapping.phone]?.trim() || '',
      address: row[mapping.address]?.trim() || '',
    })).filter(c => c.code && c.name);
    setCustomers(mapped);
    setStep('preview');
  };

  const handleImport = async () => {
    setStep('importing');
    try {
      const res = await fetch('/api/customers/import', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ customers }),
      });
      const result = await res.json();
      setImportResult({ imported: result.imported || 0, skipped: result.skipped || 0 });
      setStep('done');
    } catch {
      setImportResult({ imported: 0, skipped: customers.length });
      setStep('done');
    }
  };

  return (
    <div className="max-w-2xl mx-auto py-8">
      <h1 className="text-2xl font-bold mb-6">Nhập dữ liệu khách hàng</h1>
      
      {step === 'upload' && (
        <div className="bg-white rounded-lg border p-6">
          <h2 className="text-lg font-medium mb-4">Bước 1: Tải lên file CSV</h2>
          <p className="text-gray-600 mb-4">File CSV cần có các cột: Mã khách, Tên khách, Số điện thoại, Địa chỉ</p>
          <input type="file" accept=".csv" onChange={handleFileUpload} className="block w-full text-sm" />
        </div>
      )}

      {step === 'mapping' && (
        <div className="bg-white rounded-lg border p-6">
          <h2 className="text-lg font-medium mb-4">Bước 2: Map cột dữ liệu</h2>
          {['Mã khách', 'Tên khách', 'Số điện thoại', 'Địa chỉ'].map((field, idx) => (
            <div key={field} className="flex items-center gap-4 mb-3">
              <span className="w-32">{field}:</span>
              <select 
                value={mapping[['code', 'name', 'phone', 'address'][idx] as keyof typeof mapping]}
                onChange={e => setMapping({ ...mapping, [['code', 'name', 'phone', 'address'][idx] as keyof typeof mapping]: parseInt(e.target.value) })}
                className="border rounded px-2 py-1"
              >
                {headers.map((h, i) => <option key={i} value={i}>{h}</option>)}
              </select>
            </div>
          ))}
          <button onClick={handleMapping} className="mt-4 px-4 py-2 bg-blue-600 text-white rounded">Tiếp tục</button>
        </div>
      )}

      {step === 'preview' && (
        <div className="bg-white rounded-lg border p-6">
          <h2 className="text-lg font-medium mb-4">Bước 3: Xác nhận ({customers.length} khách hàng)</h2>
          <table className="w-full text-sm mb-4">
            <thead>
              <tr className="border-b"><th>Mã</th><th>Tên</th><th>Điện thoại</th></tr>
            </thead>
            <tbody>
              {customers.slice(0, 5).map((c, i) => (
                <tr key={i} className="border-b"><td>{c.code}</td><td>{c.name}</td><td>{c.phone}</td></tr>
              ))}
            </tbody>
          </table>
          {customers.length > 5 && <p className="text-gray-500 text-sm">... và {customers.length - 5} khách hàng nữa</p>}
          <button onClick={handleImport} className="mt-4 px-4 py-2 bg-green-600 text-white rounded">Nhập dữ liệu</button>
        </div>
      )}

      {step === 'importing' && <div className="text-center py-8">Đang nhập dữ liệu...</div>}

      {step === 'done' && importResult && (
        <div className="bg-white rounded-lg border p-6">
          <h2 className="text-lg font-medium mb-4">Hoàn tất!</h2>
          <p>Đã nhập: <strong>{importResult.imported}</strong> khách hàng</p>
          <p>Đã bỏ qua: <strong>{importResult.skipped}</strong> khách hàng</p>
          <button onClick={() => router.push('/customers')} className="mt-4 px-4 py-2 bg-blue-600 text-white rounded">
```

```typescript
import { NextRequest, NextResponse } from 'next/server';
import { PrismaClient } from '@prisma/client';
import { getServerSession } from '@/lib/auth';

const prisma = new PrismaClient();

export async function POST(req: NextRequest) {
  try {
    const session = await getServerSession();
    if (!session) {
      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
    }
    if (session.role !== 'ADMIN' && session.role !== 'ACCOUNTANT') {
      return NextResponse.json({ error: 'Forbidden' }, { status: 403 });
    }

    const { customers } = await req.json();

    if (!Array.isArray(customers)) {
      return NextResponse.json({ error: 'customers must be an array' }, { status: 400 });
    }

    const errors: string[] = [];
    let imported = 0;
    let skipped = 0;

    for (const customer of customers) {
      if (!customer.code || !customer.name) {
        skipped++;
        continue;
      }

      try {
        // Check if customer code already exists for this factory
        const existing = await prisma.customer.findFirst({
          where: { code: customer.code, factoryId: session.factoryId! },
        });

        if (existing) {
          skipped++;
          continue;
        }

        await prisma.customer.create({
          data: {
            code: customer.code,
            name: customer.name,
            phone: customer.phone || null,
            address: customer.address || null,
            factoryId: session.factoryId!,
          },
        });
        imported++;
      } catch (err) {
        errors.push(`Error importing ${customer.code}: ${err}`);
        skipped++;
      }
    }

    return NextResponse.json({ imported, skipped, errors });
  } catch (error) {
    console.error('Customer import error:', error);
    return NextResponse.json({ error: 'Internal server error' }, { status: 500 });
  }
}
```
