'use client';

import { useState } from 'react';
import type { Subscription } from '@prisma/client';
import { exportCustomerDebtExcel } from '@/lib/export-excel';
import { useRouter } from 'next/navigation';

interface CancelSubscriptionFlowProps {
  subscription: Subscription;
  factoryId: string;
}

export function CancelSubscriptionFlow({ subscription, factoryId }: CancelSubscriptionFlowProps) {
  const [showModal, setShowModal] = useState(false);
  const [cancelling, setCancelling] = useState(false);
  const router = useRouter();

  const handleExportAndCancel = async () => {
    setCancelling(true);
    
    try {
      // 1. Fetch customer and order data for export
      const [customersRes, ordersRes] = await Promise.all([
        fetch(`/api/customers?factoryId=${factoryId}`),
        fetch(`/api/orders?factoryId=${factoryId}&months=6`),
      ]);
      
      const { customers } = await customersRes.json();
      const { orders } = await ordersRes.json();

      // 2. Trigger Excel download immediately (before cancel)
      if (customers && orders) {
        exportCustomerDebtExcel(customers, orders);
      }

      // 3. Call cancel API
      const cancelRes = await fetch('/api/subscriptions/cancel', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ subscriptionId: subscription.id }),
      });

      if (cancelRes.ok) {
        router.refresh();
        setShowModal(false);
        alert('Đã hủy dịch vụ. Dữ liệu đã được xuất file Excel.');
      } else {
        alert('Có lỗi xảy ra khi hủy dịch vụ.');
      }
    } catch (error) {
      console.error('Cancel error:', error);
      alert('Có lỗi xảy ra khi hủy dịch vụ.');
    } finally {
      setCancelling(false);
    }
  };

  const handleContinueTrial = () => {
    setShowModal(false);
  };

  const handleContactSupport = () => {
    window.location.href = 'mailto:support@iceflow.vn';
  };

  return (
    <>
      <button
        onClick={() => setShowModal(true)}
        className="text-red-600 text-sm underline hover:no-underline"
      >
        Hủy dịch vụ
      </button>

      {showModal && (
        <div className="fixed inset-0 z-50 flex items-center justify-center bg-black/50">
          <div className="bg-white rounded-lg shadow-xl max-w-md w-full mx-4 p-6">
            <h2 className="text-lg font-semibold mb-4">Hủy dịch vụ</h2>
            <p className="text-gray-600 mb-6">
              Trước khi xác nhận hủy, bạn có muốn xuất dữ liệu công nợ không? 
              Dữ liệu sẽ được tải về dưới dạng file Excel.
            </p>
            
            <div className="flex flex-col gap-3">
              <button
                onClick={handleExportAndCancel}
                disabled={cancelling}
                className="w-full px-4 py-2 bg-red-600 text-white rounded-lg font-medium hover:bg-red-700 disabled:opacity-50"
              >
                {cancelling ? 'Đang xử lý...' : 'Xuất dữ liệu & Hủy'}
              </button>
              <button
                onClick={handleContinueTrial}
                className="w-full px-4 py-2 bg-gray-100 text-gray-700 rounded-lg font-medium hover:bg-gray-200"
              >
                Tiếp tục dùng thử
              </button>

                className="w-full px-4 py-2 text-gray-500 underline hover:no-underline text-sm"

                Liên hệ hỗ trợ

```typescript
import * as XLSX from 'xlsx';

// Types for the export
interface CustomerWithDebt {
  code: string;
  name: string;
  phone?: string | null;
  address?: string | null;
  debtLedger?: Array<{ amount: number; type: string }>;
  shellDeposit?: { shellsOut: number; depositPerShell: number };
}

interface Order {
  id: string;
  createdAt: string | Date;
  total: number;
  status: string;
  customer?: { name: string };
}

interface ExportData {
  customers: CustomerWithDebt[];
  orders: Order[];
}

export function exportCustomerDebtExcel(customers: CustomerWithDebt[], orders: Order[]): void {
  const wb = XLSX.utils.book_new();

  // Sheet 1: Customer list with debt
  const sheet1Data = customers.map(c => {
    // Calculate total debt from ledger (DEBT type adds, PAYMENT type subtracts)
    const totalDebt = (c.debtLedger || []).reduce((sum, entry) => {
      if (entry.type === 'DEBT') return sum + Number(entry.amount);
      if (entry.type === 'PAYMENT') return sum - Number(entry.amount);
      return sum;
    }, 0);
    
    // Calculate shell deposit owed
    const shellDeposit = c.shellDeposit 
      ? Number(c.shellDeposit.shellsOut) * Number(c.shellDeposit.depositPerShell)
      : 0;

    return {
      'Mã khách hàng': c.code,
      'Tên khách hàng': c.name,
      'Số điện thoại': c.phone || '',
      'Địa chỉ': c.address || '',
      'Công nợ': totalDebt,
      'Tiền đặt cọc vỏ': shellDeposit,
    };
  });

  const ws1 = XLSX.utils.json_to_sheet(sheet1Data);
  XLSX.utils.book_append_sheet(wb, ws1, 'Cong No');

  // Sheet 2: Order history (last 3-6 months)
  const sheet2Data = orders.map(o => ({
    'Mã đơn': o.id,
    'Khách hàng': o.customer?.name || '',
    'Ngày tạo': o.createdAt instanceof Date ? o.createdAt.toISOString().split('T')[0] : o.createdAt,
    'Tổng tiền': o.total,
    'Trạng thái': o.status,
  }));

  const ws2 = XLSX.utils.json_to_sheet(sheet2Data);
  XLSX.utils.book_append_sheet(wb, ws2, 'Lich Su Don Hang');

  // Trigger browser download
  XLSX.writeFile(wb, 'iceflow-export.xlsx');
}
```

Key behaviors:
- Sheet 1: Mã khách hàng, Tên khách hàng, Số điện thoại, Địa chỉ, Công nợ, Tiền đặt cọc vỏ
- Sheet 2: Mã đơn, Khách hàng, Ngày tạo, Tổng tiền, Trạng thái
- Debt calculation: DEBT entries add to debt, PAYMENT entries subtract
- Shell deposit: shellsOut * depositPerShell
- XLSX.writeFile triggers browser download of iceflow-export.xlsx

```typescript
import { NextRequest, NextResponse } from 'next/server';
import { PrismaClient } from '@prisma/client';
import { getServerSession } from '@/lib/auth';

const prisma = new PrismaClient();

export async function POST(req: NextRequest) {
  try {
    const session = await getServerSession();
    if (!session) {
      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
    }
    if (session.role !== 'ADMIN') {
      return NextResponse.json({ error: 'Admin only' }, { status: 403 });
    }

    // Defense-in-depth: check subscription still active before allowing cancel
    const { requireWritableSubscription } = await import('@/lib/subscription-status');
    const writableCheck = await requireWritableSubscription(session.factoryId);
    if (!writableCheck.allowed) {
      return NextResponse.json({ error: writableCheck.error }, { status: 403 });
    }

    const { subscriptionId } = await req.json();

    if (!subscriptionId) {
      return NextResponse.json({ error: 'subscriptionId required' }, { status: 400 });
    }

    // Verify subscription belongs to this factory
    const subscription = await prisma.subscription.findUnique({
      where: { id: subscriptionId },
    });

    if (!subscription || subscription.factoryId !== session.factoryId) {
      return NextResponse.json({ error: 'Subscription not found' }, { status: 404 });
    }

    // Set grace period using gracePeriodEnd field, PRESERVE original endDate for billing history
    const gracePeriodEnd = new Date(Date.now() + 7 * 24 * 60 * 60 * 1000);

    const updatedSubscription = await prisma.subscription.update({
      where: { id: subscriptionId },
      data: {
        status: 'CANCELLED',
        gracePeriodEnd: gracePeriodEnd,
        // NOTE: endDate is PRESERVED (not overwritten) so billing history remains intact
      },
    });
      success: true,
      gracePeriodEnd: gracePeriodEnd.toISOString(),
      subscription: updatedSubscription,
    });
  } catch (error) {
    console.error('Cancel subscription error:', error);
    return NextResponse.json({ error: 'Internal server error' }, { status: 500 });
  }
}
```

```typescript
import { NextRequest, NextResponse } from 'next/server';
import { PrismaClient } from '@prisma/client';
import { getServerSession } from '@/lib/auth';

const prisma = new PrismaClient();

export async function GET(req: NextRequest, { params }: { params: { id: string } }) {
  const session = await getServerSession();
  if (!session) {
    return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
  }

  const subscription = await prisma.subscription.findUnique({
    where: { id: params.id },
  });

  if (!subscription || subscription.factoryId !== session.factoryId) {
    return NextResponse.json({ error: 'Not found' }, { status: 404 });
  }

  return NextResponse.json({ subscription });
}

export async function PATCH(req: NextRequest, { params }: { params: { id: string } }) {
  const session = await getServerSession();
  if (!session) {
    return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
  }
  if (session.role !== 'ADMIN') {
    return NextResponse.json({ error: 'Admin only' }, { status: 403 });
  }

  // Defense-in-depth: check subscription still active before allowing update
  const { requireWritableSubscription } = await import('@/lib/subscription-status');
  const writableCheck = await requireWritableSubscription(session.factoryId);
  if (!writableCheck.allowed) {
    return NextResponse.json({ error: writableCheck.error }, { status: 403 });
  }

  const subscription = await prisma.subscription.findUnique({
    where: { id: params.id },
  });

  if (!subscription || subscription.factoryId !== session.factoryId) {
    return NextResponse.json({ error: 'Not found' }, { status: 404 });
  }

  const body = await req.json();

  // Compute actual driver count from Driver table (NOT from client-supplied body.driverCount)
  // Client-supplied driverCount is ignored — we trust the actual database count
  const actualDriverCount = await prisma.driver.count({
    where: { factoryId: session.factoryId, isActive: true },
  });

  // Only allow updating pricePerDriver (driverCount is always derived from Driver table)
  const updates: { pricePerDriver?: number } = {};

  if (typeof body.pricePerDriver === 'number' && body.pricePerDriver > 0) {
    updates.pricePerDriver = body.pricePerDriver;
  }

  const updated = await prisma.subscription.update({
    where: { id: params.id },
    data: {
      ...updates,
      driverCount: actualDriverCount, // Always use computed count
    },
  });

  return NextResponse.json({ subscription: updated });
}
```
