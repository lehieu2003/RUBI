Create `src/app/api/factory/onboarding/route.ts` as a new file:

```typescript
import { NextRequest, NextResponse } from 'next/server';
import { prisma } from '@/lib/db';
import { requireRole } from '@/lib/auth/helpers';

export async function PATCH(req: NextRequest) {
  try {
    const session = await requireRole(['ADMIN']);
    if (!session.factoryId) {
      return NextResponse.json({ error: 'Bad Request: factoryId not found' }, { status: 400 });
    }

    await prisma.factory.update({
      where: { id: session.factoryId },
      data: { onboardingCompleted: true },
    });

    return NextResponse.json({ success: true });
  } catch (error) {
    if (error instanceof Response) throw error; // Re-throw NextResponse from requireRole
    console.error('Onboarding complete error:', error);
    return NextResponse.json({ error: 'Lỗi server' }, { status: 500 });
  }
}
```

Key points:
- Uses `requireRole(['ADMIN'])` to ensure only ADMIN users can mark onboarding complete
- factoryId comes from session (set during login/register)
- Updates `onboardingCompleted: true` on the Factory
- Throws if requireRole returns a Response (propagates auth errors)
- Generic error message on failure, no stack traces
