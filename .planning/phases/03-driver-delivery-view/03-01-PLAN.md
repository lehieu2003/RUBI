Create directory src/app/api/driver/orders/ and file route.ts.

The route authenticates driver by token URL param, fetches orders for today where status is PRINTED or DELIVERED and driverId matches.

```typescript
import { NextRequest, NextResponse } from 'next/server';
import { prisma } from '@/lib/db';

// GET /api/driver/orders?token=xxx&date=YYYY-MM-DD
// Returns orders for authenticated driver filtered by today's date
// Orders visible when status IN (PRINTED, DELIVERED)
export async function GET(request: NextRequest) {
  try {
    const { searchParams } = new URL(request.url);
    const token = searchParams.get('token');
    const dateParam = searchParams.get('date');

    if (!token) {
      return NextResponse.json({ error: 'Token required' }, { status: 401 });
    }

    // Look up driver by token
    const driver = await prisma.driver.findUnique({
      where: { token },
    });

    if (!driver) {
      return NextResponse.json({ error: 'Invalid token' }, { status: 401 });
    }

    // Get today's date range
    const date = dateParam ? new Date(dateParam) : new Date();
    const startOfDay = new Date(date);
    startOfDay.setHours(0, 0, 0, 0);
    const endOfDay = new Date(date);
    endOfDay.setHours(23, 59, 59, 999);

    // Fetch orders for this driver with PRINTED or DELIVERED status
    // Note: Using 'PRINTED' and 'DELIVERED' as per Phase 3 D-06
    const orders = await prisma.order.findMany({
      where: {
        driverId: driver.id,
        status: { in: ['PRINTED', 'DELIVERED'] },
        createdAt: { gte: startOfDay, lte: endOfDay },
      },
      include: {
        customer: true,
        orderItems: { include: { product: true } },
      },
      orderBy: { createdAt: 'asc' },
    });

    // Serialize Decimals and add driver name to response
    const serialized = orders.map((order) => ({
      ...order,
      total: order.total.toNumber(),
      orderItems: order.orderItems.map((item) => ({
        ...item,
        unitPrice: item.unitPrice.toNumber(),
        lineTotal: item.lineTotal.toNumber(),
      })),
      driverName: driver.name, // Include driver name for display
    }));

    return NextResponse.json({
      driver: { id: driver.id, name: driver.name },
      orders: serialized,
    });
  } catch (error) {
    console.error('[GET /api/driver/orders]', error);
    return NextResponse.json({ error: 'Failed to fetch orders' }, { status: 500 });
  }
}
```

```typescript
import { NextRequest, NextResponse } from 'next/server';
import { prisma } from '@/lib/db';
import { z } from 'zod';

// PATCH /api/orders/[id]/confirm
// Body: { gps: { lat: number, lng: number } | null }
// Updates status to DELIVERED and saves GPS coordinates
const ConfirmSchema = z.object({
  gps: z.object({
    lat: z.number(),
    lng: z.number(),
  }).nullable().optional(), // Can be null if GPS capture failed (D-09)
});

export async function PATCH(
  request: NextRequest,
  { params }: { params: Promise<{ id: string }> }
) {
  try {
    const { id } = await params;
    const body = await request.json();
    const parsed = ConfirmSchema.safeParse(body);

    if (!parsed.success) {
      return NextResponse.json(
        { error: 'Invalid request. Required: gps { lat, lng } or gps: null' },
        { status: 400 }
      );
    }

    // Update order to DELIVERED with GPS coordinates
    const order = await prisma.order.update({
      where: { id },
      data: {
        status: 'DELIVERED',
        deliveryGps: parsed.data.gps, // Json field - {lat, lng} or null
      },
    });

    return NextResponse.json({
      success: true,
      orderId: order.id,
      status: order.status,
      deliveryGps: order.deliveryGps,
    });
  } catch (error) {
    console.error('[PATCH /api/orders/[id]/confirm]', error);
    return NextResponse.json({ error: 'Confirmation failed' }, { status: 500 });
  }
}
```

```typescript
import { NextRequest, NextResponse } from 'next/server';
import { prisma } from '@/lib/db';
import { z } from 'zod';

// PATCH /api/orders/bulk-assign
// Body: { orderIds: string[], driverId: string }
// Assigns multiple orders to a single driver
const BulkAssignSchema = z.object({
  orderIds: z.array(z.string()).min(1),
  driverId: z.string(),
});

export async function PATCH(request: NextRequest) {
  try {
    const body = await request.json();
    const parsed = BulkAssignSchema.safeParse(body);

    if (!parsed.success) {
      return NextResponse.json(
        { error: 'Invalid request. Required: orderIds (string[]) and driverId (string)' },
        { status: 400 }
      );
    }

    const { orderIds, driverId } = parsed.data;

    // Verify driver exists
    const driver = await prisma.driver.findUnique({
      where: { id: driverId },
    });

    if (!driver) {
      return NextResponse.json({ error: 'Driver not found' }, { status: 404 });
    }

    // Bulk update orders to assign to driver
    // Only updates orders that are in PRINTED status (D-06: order appears when PRINTED AND driverId IS NOT NULL)
    const result = await prisma.order.updateMany({
      where: {
        id: { in: orderIds },
        status: 'PRINTED', // Only assign to PRINTED orders
      },
      data: {
        driverId: driverId,
      },
    });

    return NextResponse.json({
      success: true,
      assignedCount: result.count,
      driver: { id: driver.id, name: driver.name },
    });
  } catch (error) {
    console.error('[PATCH /api/orders/bulk-assign]', error);
    return NextResponse.json({ error: 'Bulk assign failed' }, { status: 500 });
  }
}
```

This endpoint returns all orders with GPS data for the admin map view (D-21, D-22, D-23).

```typescript
import { NextRequest, NextResponse } from 'next/server';
import { prisma } from '@/lib/db';

// GET /api/orders/map?date=YYYY-MM-DD&driverId=xxx
// Returns all orders with GPS data for admin map
// Supports filtering by date and/or driver
export async function GET(request: NextRequest) {
  try {
    const { searchParams } = new URL(request.url);
    const dateParam = searchParams.get('date');
    const driverId = searchParams.get('driverId');

    // Build date range
    const date = dateParam ? new Date(dateParam) : new Date();
    const startOfDay = new Date(date);
    startOfDay.setHours(0, 0, 0, 0);
    const endOfDay = new Date(date);
    endOfDay.setHours(23, 59, 59, 999);

    // Build where clause
    const where: Record<string, unknown> = {
      status: { in: ['PRINTED', 'DELIVERED'] },
      createdAt: { gte: startOfDay, lte: endOfDay },
      deliveryGps: { not: null }, // Only orders with GPS data
    };

    if (driverId) {
      where.driverId = driverId;
    }

    const orders = await prisma.order.findMany({
      where,
      include: {
        customer: true,
        driver: { select: { id: true, name: true } },
      },
      orderBy: { createdAt: 'asc' },
    });

    // Serialize for client
    const serialized = orders.map((order) => ({
      id: order.id,
      status: order.status,
      customerName: order.customer.name,
      customerAddress: order.customer.address,
      customerPhone: order.customer.phone,
      deliveryGps: order.deliveryGps, // Json field {lat, lng}
      driverName: order.driver?.name || null,
      driverId: order.driverId,
      createdAt: order.createdAt,
    }));

    return NextResponse.json(serialized);
  } catch (error) {
    console.error('[GET /api/orders/map]', error);
    return NextResponse.json({ error: 'Failed to fetch map data' }, { status: 500 });
  }
}
```
